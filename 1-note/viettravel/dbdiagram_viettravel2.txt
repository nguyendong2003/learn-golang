Enum user_status_enum {
  ACTIVE
  LOCKED
  DELETED
}

Table roles {
  id uuid [primary key, default: `gen_random_uuid()`]
  name varchar(100) [not null, unique]
  description text
  
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
  deleted_at timestamp
}

Table permissions {
  id uuid [primary key, default: `gen_random_uuid()`]
  name varchar(100) [not null]
  code varchar(100) [unique, not null, note: 'Example: agent.approve, tour.create']
  description text
  
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
  deleted_at timestamp
}

Table users {
  id uuid [primary key, default: `gen_random_uuid()`]
  role_id uuid [ref: > roles.id, not null]
  
  email varchar(255) [unique, not null]
  username varchar(255) [unique, not null]
  password varchar(255) [not null]
  name varchar(255)
  phone varchar(20) [unique]
  avatar_url text
  
  status user_status_enum [default: 'ACTIVE']
  is_affiliate boolean [default: false]
    
  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
  deleted_at timestamp
}

Table role_permissions {
  role_id uuid [ref: > roles.id, not null]
  permission_id uuid [ref: > permissions.id, not null]
  
  indexes {
    (role_id, permission_id) [pk]
  }
}

Table user_roles {
  user_id uuid [ref: > users.id, not null]
  role_id uuid [ref: > roles.id, not null]
  
  indexes {
    (user_id, role_id) [pk]
  }
}

  id uuid [primary key, default: `gen_random_uuid()`]

  representative_name varchar(255) [note: 'Tên người đại diện']
  representative_id_number varchar(20) [unique, note: 'Số CCCD']
  
  // 2. Nhóm thông tin giới thiệu & truyền thông (BR-61: Auto update, không cần duyệt)

  established_year integer [note: 'Năm thành lập']
  company_size varchar(50) [note: 'Quy mô nhân sự (Ví dụ: 10-50 nhân viên)']
  working_hours varchar(255) [note: 'Giờ làm việc (Ví dụ: 08:00 - 17:00)']

  facebook_url varchar(255)
  zalo_url varchar(255)

  status_note text [note: 'Lý do từ chối, yêu cầu sửa đổi hồ sơ, hoặc lý do khóa/cấm tài khoản']
