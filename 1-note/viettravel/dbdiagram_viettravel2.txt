  WEEKEND      [note: 'Giá cuối tuần (Thứ 6, 7, CN)']

// 1. Bảng Cấu hình Giá & Phụ thu cơ bản (Quan hệ 1:1 với loại phòng)
Table room_price_settings {
  id uuid [primary key, default: `gen_random_uuid()`]
  room_type_id uuid [ref: > room_types.id, not null, note: 'Liên kết 1:1 với loại phòng']
  
  // Giá cơ bản khi không dính luật đặc biệt nào
  base_price decimal(15, 2) [not null, note: 'Giá phòng mặc định/đêm (Cấu hình tại tab Giá)']
  
  // Các khoản phụ thu cố định
  extra_adult_fee decimal(15, 2) [default: 0, note: 'Phụ thu thêm người lớn quá số lượng tiêu chuẩn']
  extra_bed_fee decimal(15, 2) [default: 0, note: 'Phụ thu tiền thêm giường phụ (Extra bed)']
  breakfast_fee decimal(15, 2) [default: 0, note: 'Phụ thu tiền ăn sáng bán lẻ tính theo người/đêm']
  
  // Phụ phí trẻ em 6-12 tuổi (phụ phí) (extra_child_fee = extra_adult_fee * (1 - extra_child_fee_percent) / 100.0)
  extra_child_fee_percent decimal(5, 2) [default: 0, note: 'Phần trăm giá người lớn áp dụng cho trẻ em']
  
  is_active boolean [default: true, not null]

  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
  deleted_at timestamp

  // Partial Index
  indexes {
    room_type_id [
      name: "idx_room_price_settings_type_unique_active", 
      note: "Partial Unique Index: Chỉ UNIQUE khi WHERE deleted_at IS NULL"
    ]
  }
}

// 2. Luật điều chỉnh giá linh hoạt theo thời điểm
Table room_price_rules {
  id uuid [primary key, default: `gen_random_uuid()`]
  room_type_id uuid [ref: > room_types.id, not null, note: 'Áp dụng cho loại phòng nào']
  
  name varchar(255) [not null, note: 'Tên đợt cấu hình (VD: Tết Nguyên Đán 2026, Mùa hè cao điểm)']
  rate_type rate_type_enum [not null, note: 'Thứ tự ưu tiên tính toán: PROMOTIONAL > HOLIDAY > WEEKEND > SEASONAL']
  
  // Áp dụng cho cấu hình theo khoảng ngày (Mùa, Lễ, Khuyến mãi)
  start_date date [note: 'Ngày bắt đầu áp dụng luật (Bỏ trống nếu là cấu hình cuối tuần cố định)']
  end_date date [note: 'Ngày kết thúc áp dụng luật']
  
  // Áp dụng riêng cho cấu hình Cuối tuần (Lưu mảng JSON số ngày trong tuần. VD: [4, 5, 6] tương ứng Thứ 6, Thứ 7, Chủ Nhật)
  days_of_week integer[] [note: 'Mảng các ngày trong tuần áp dụng cho rule WEEKEND']
  
  // Số tiền thực tế sẽ bán cho mỗi đêm khi dính luật này
  price_value decimal(15, 2) [not null, note: 'Mức giá phòng áp dụng thực tế cho mỗi đêm']
  
  is_active boolean [default: true, not null]

  created_at timestamp [default: `now()`]
  updated_at timestamp [default: `now()`]
  deleted_at timestamp

  indexes {
    (room_type_id, rate_type)
    (start_date, end_date)
    // Partial index để check các rule đang active hiệu quả hơn
    (room_type_id, start_date, end_date) [name: "idx_price_rules_perf", note: "Tăng tốc độ quét luật giá theo ngày"]
  }
}
